cmake_minimum_required(VERSION 3.20)
project(OGSimNet CXX)

if(PROJECT_IS_TOP_LEVEL)
    set(CMAKE_CXX_STANDARD 20)
    set(CMAKE_CXX_STANDARD_REQUIRED ON)
    if(MSVC)
        add_compile_options(/Zc:preprocessor /utf-8)
    endif()
    add_compile_definitions(OG_STANDALONE_BUILD)
    enable_testing()
endif()

if(NOT TARGET og_simulation)
    set(OG_SIMULATION_DIR "" CACHE PATH "Root of an og-simulation checkout (the directory that holds OGSimulation/ and glm/)")
    if(NOT EXISTS "${OG_SIMULATION_DIR}/OGSimulation/CMakeLists.txt")
        message(FATAL_ERROR "og-sim-net: set OG_SIMULATION_DIR to the root of an og-simulation checkout")
    endif()
    if(NOT TARGET glm::glm-header-only)
        add_subdirectory("${OG_SIMULATION_DIR}/glm" "${CMAKE_CURRENT_BINARY_DIR}/glm")
    endif()
    add_subdirectory("${OG_SIMULATION_DIR}/OGSimulation" "${CMAKE_CURRENT_BINARY_DIR}/og_simulation")
endif()

if(NOT TARGET og_net)
    set(OG_NET_DIR "" CACHE PATH "Root of an og-net checkout (the directory that holds OGNet/ and tests/)")
    if(NOT EXISTS "${OG_NET_DIR}/OGNet")
        message(FATAL_ERROR "og-sim-net: set OG_NET_DIR to the root of an og-net checkout")
    endif()
    add_subdirectory("${OG_NET_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/og_net")
endif()

file(GLOB_RECURSE OGSIMNET_HEADERS CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/OGSimNet/*.h")

add_library(og_sim_net INTERFACE)
target_sources(og_sim_net INTERFACE ${OGSIMNET_HEADERS})
target_include_directories(og_sim_net INTERFACE ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_features(og_sim_net INTERFACE cxx_std_20)
target_link_libraries(og_sim_net INTERFACE og_simulation og_net)
if(CMAKE_CXX_COMPILER_ID STREQUAL "MSVC")
    target_compile_options(og_sim_net INTERFACE /fp:precise)
elseif(CMAKE_CXX_COMPILER_FRONTEND_VARIANT STREQUAL "MSVC")
    target_compile_options(og_sim_net INTERFACE /fp:precise /clang:-ffp-contract=off)
else()
    target_compile_options(og_sim_net INTERFACE -ffp-contract=off)
endif()

set(OGSIMNET_HEADER_TUS "")
foreach(header IN LISTS OGSIMNET_HEADERS)
    file(RELATIVE_PATH header_rel "${CMAKE_CURRENT_SOURCE_DIR}" "${header}")
    string(MAKE_C_IDENTIFIER "${header_rel}" header_id)
    set(header_tu "${CMAKE_CURRENT_BINARY_DIR}/header_check/${header_id}.cpp")
    set(header_tu_text "#include \"${header_rel}\"\n")
    if(EXISTS "${header_tu}")
        file(READ "${header_tu}" header_tu_old)
    else()
        set(header_tu_old "")
    endif()
    if(NOT header_tu_old STREQUAL header_tu_text)
        file(WRITE "${header_tu}" "${header_tu_text}")
    endif()
    list(APPEND OGSIMNET_HEADER_TUS "${header_tu}")
endforeach()

add_library(og_sim_net_header_check OBJECT ${OGSIMNET_HEADER_TUS})
target_link_libraries(og_sim_net_header_check PRIVATE og_sim_net)
if(MSVC)
    target_compile_options(og_sim_net_header_check PRIVATE /GR- /EHs-c- /wd4577)
    target_compile_definitions(og_sim_net_header_check PRIVATE _HAS_EXCEPTIONS=0)
else()
    target_compile_options(og_sim_net_header_check PRIVATE -fno-rtti -fno-exceptions)
endif()

set(OG_SIMULATION_TESTS_DIR "" CACHE PATH "Root of an og-simulation-tests checkout (the directory that holds Source/OGSimulationTests/)")
set(OGNET_CATCH2_DIR "" CACHE PATH "Directory holding catch_amalgamated.hpp and catch_amalgamated.cpp (Catch2 v3); empty skips the tests")

set(OGSIMNET_TESTS_DIR "${OG_SIMULATION_TESTS_DIR}/Source/OGSimulationTests")
if(OGNET_CATCH2_DIR STREQUAL "" OR OG_SIMULATION_TESTS_DIR STREQUAL "")
    message(STATUS "og-sim-net: OGNET_CATCH2_DIR or OG_SIMULATION_TESTS_DIR is empty; og_sim_net_tests is not built")
elseif(NOT EXISTS "${OGNET_CATCH2_DIR}/catch_amalgamated.hpp" OR NOT EXISTS "${OGNET_CATCH2_DIR}/catch_amalgamated.cpp")
    message(FATAL_ERROR "og-sim-net: OGNET_CATCH2_DIR (${OGNET_CATCH2_DIR}) holds no catch_amalgamated.hpp and catch_amalgamated.cpp")
elseif(NOT EXISTS "${OGSIMNET_TESTS_DIR}/OgTagAliases.cpp")
    message(FATAL_ERROR "og-sim-net: OG_SIMULATION_TESTS_DIR (${OG_SIMULATION_TESTS_DIR}) holds no Source/OGSimulationTests/OgTagAliases.cpp")
else()
    if(NOT TARGET Catch2)
        add_library(Catch2 STATIC "${OGNET_CATCH2_DIR}/catch_amalgamated.cpp")
        target_include_directories(Catch2 PUBLIC "${OGNET_CATCH2_DIR}")
    endif()
    get_target_property(OGSIMNET_OGNET_ROOT og_net SOURCE_DIR)
    if(NOT TARGET og_tickdriver)
        get_target_property(OGSIMNET_OGSIM_ROOT og_simulation SOURCE_DIR)
        add_subdirectory("${OGSIMNET_OGSIM_ROOT}/../og-tickdriver" "${CMAKE_CURRENT_BINARY_DIR}/og_tickdriver")
    endif()
    file(GLOB OGSIMNET_TEST_SOURCES CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/tests/*.cpp")
    add_executable(og_sim_net_tests ${OGSIMNET_TEST_SOURCES} "${OGSIMNET_TESTS_DIR}/OgTagAliases.cpp")
    target_include_directories(og_sim_net_tests PRIVATE "${OGSIMNET_OGNET_ROOT}/tests")
    target_compile_definitions(og_sim_net_tests PRIVATE WITH_LOW_LEVEL_TESTS=1)
    target_link_libraries(og_sim_net_tests PRIVATE og_sim_net og_tickdriver Catch2)
    if(MSVC)
        target_compile_options(og_sim_net_tests PRIVATE /bigobj /GR-)
    else()
        target_compile_options(og_sim_net_tests PRIVATE -fno-rtti)
    endif()
    if(ANDROID)
        target_link_libraries(og_sim_net_tests PRIVATE log)
    endif()
    if(EXISTS "${OGNET_CATCH2_DIR}/Catch.cmake")
        list(APPEND CMAKE_MODULE_PATH "${OGNET_CATCH2_DIR}")
        include(Catch)
        catch_discover_tests(og_sim_net_tests TEST_SPEC "[@og]" TEST_PREFIX "og_sim_net: " PROPERTIES LABELS og_sim_net)
    else()
        add_test(NAME og_sim_net_tests COMMAND og_sim_net_tests "[@og]")
        set_tests_properties(og_sim_net_tests PROPERTIES LABELS og_sim_net)
    endif()
endif()
