cmake_minimum_required(VERSION 3.20)
project(OGSimSession CXX)

if(PROJECT_IS_TOP_LEVEL)
    set(CMAKE_CXX_STANDARD 20)
    set(CMAKE_CXX_STANDARD_REQUIRED ON)
    if(MSVC)
        add_compile_options(/Zc:preprocessor /utf-8)
    endif()
    add_compile_definitions(OG_STANDALONE_BUILD)
    enable_testing()
endif()

if(NOT TARGET og_sim_net)
    set(OG_SIM_NET_DIR "" CACHE PATH "Root of an og-sim-net folder (the directory that holds OGSimNet/ and its CMakeLists.txt)")
    if(NOT EXISTS "${OG_SIM_NET_DIR}/CMakeLists.txt")
        message(FATAL_ERROR "og-sim-session: set OG_SIM_NET_DIR to the og-sim-net folder (it takes og-simulation and og-net the same way)")
    endif()
    add_subdirectory("${OG_SIM_NET_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/og_sim_net")
endif()

if(NOT TARGET og_simulation_jolt)
    set(OG_SIMULATION_JOLT_DIR "" CACHE PATH "Root of an og-simulation-jolt checkout; empty builds the session without it")
    if(NOT OG_SIMULATION_JOLT_DIR STREQUAL "")
        if(NOT EXISTS "${OG_SIMULATION_JOLT_DIR}/OGSimulationJolt/CMakeLists.txt")
            message(FATAL_ERROR "og-sim-session: OG_SIMULATION_JOLT_DIR (${OG_SIMULATION_JOLT_DIR}) is not an og-simulation-jolt checkout")
        endif()
        add_subdirectory("${OG_SIMULATION_JOLT_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/og_simulation_jolt")
    endif()
endif()

if(NOT TARGET og_tickdriver)
    get_target_property(OGSIMSESSION_OGSIM_ROOT og_simulation SOURCE_DIR)
    add_subdirectory("${OGSIMSESSION_OGSIM_ROOT}/../og-tickdriver" "${CMAKE_CURRENT_BINARY_DIR}/og_tickdriver")
endif()

file(GLOB_RECURSE OGSIMSESSION_HEADERS CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/OGSimSession/*.h")

add_library(og_sim_session INTERFACE)
target_sources(og_sim_session INTERFACE ${OGSIMSESSION_HEADERS})
target_include_directories(og_sim_session INTERFACE ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_features(og_sim_session INTERFACE cxx_std_20)
target_link_libraries(og_sim_session INTERFACE og_simulation og_tickdriver og_sim_net)
if(CMAKE_CXX_COMPILER_ID STREQUAL "MSVC")
    target_compile_options(og_sim_session INTERFACE /fp:precise)
elseif(CMAKE_CXX_COMPILER_FRONTEND_VARIANT STREQUAL "MSVC")
    target_compile_options(og_sim_session INTERFACE /fp:precise /clang:-ffp-contract=off)
else()
    target_compile_options(og_sim_session INTERFACE -ffp-contract=off)
endif()

set(OGSIMSESSION_HEADER_TUS "")
foreach(header IN LISTS OGSIMSESSION_HEADERS)
    file(RELATIVE_PATH header_rel "${CMAKE_CURRENT_SOURCE_DIR}" "${header}")
    string(MAKE_C_IDENTIFIER "${header_rel}" header_id)
    set(header_tu "${CMAKE_CURRENT_BINARY_DIR}/header_check/${header_id}.cpp")
    set(header_tu_text "#include \"${header_rel}\"\n")
    if(EXISTS "${header_tu}")
        file(READ "${header_tu}" header_tu_old)
    else()
        set(header_tu_old "")
    endif()
    if(NOT header_tu_old STREQUAL header_tu_text)
        file(WRITE "${header_tu}" "${header_tu_text}")
    endif()
    list(APPEND OGSIMSESSION_HEADER_TUS "${header_tu}")
endforeach()

add_library(og_sim_session_header_check OBJECT ${OGSIMSESSION_HEADER_TUS})
target_link_libraries(og_sim_session_header_check PRIVATE og_sim_session)
if(MSVC)
    target_compile_options(og_sim_session_header_check PRIVATE /GR- /EHs-c- /wd4577)
    target_compile_definitions(og_sim_session_header_check PRIVATE _HAS_EXCEPTIONS=0)
else()
    target_compile_options(og_sim_session_header_check PRIVATE -fno-rtti -fno-exceptions)
endif()

set(OG_SIMULATION_TESTS_DIR "" CACHE PATH "Root of an og-simulation-tests checkout (the directory that holds Source/OGSimulationTests/)")
set(OGNET_CATCH2_DIR "" CACHE PATH "Directory holding catch_amalgamated.hpp and catch_amalgamated.cpp (Catch2 v3); empty skips the tests")

set(OGSIMSESSION_TESTS_DIR "${OG_SIMULATION_TESTS_DIR}/Source/OGSimulationTests")
if(OGNET_CATCH2_DIR STREQUAL "" OR OG_SIMULATION_TESTS_DIR STREQUAL "")
    message(STATUS "og-sim-session: OGNET_CATCH2_DIR or OG_SIMULATION_TESTS_DIR is empty; og_sim_session_tests is not built")
elseif(NOT EXISTS "${OGNET_CATCH2_DIR}/catch_amalgamated.hpp" OR NOT EXISTS "${OGNET_CATCH2_DIR}/catch_amalgamated.cpp")
    message(FATAL_ERROR "og-sim-session: OGNET_CATCH2_DIR (${OGNET_CATCH2_DIR}) holds no catch_amalgamated.hpp and catch_amalgamated.cpp")
elseif(NOT EXISTS "${OGSIMSESSION_TESTS_DIR}/TickDriver/TickDriverFakes.h" OR NOT EXISTS "${OGSIMSESSION_TESTS_DIR}/OgTagAliases.cpp")
    message(FATAL_ERROR "og-sim-session: OG_SIMULATION_TESTS_DIR (${OG_SIMULATION_TESTS_DIR}) holds no Source/OGSimulationTests/TickDriver/TickDriverFakes.h and OgTagAliases.cpp")
else()
    if(NOT TARGET Catch2)
        add_library(Catch2 STATIC "${OGNET_CATCH2_DIR}/catch_amalgamated.cpp")
        target_include_directories(Catch2 PUBLIC "${OGNET_CATCH2_DIR}")
    endif()
    file(GLOB OGSIMSESSION_TEST_SOURCES CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/tests/*.cpp")
    list(REMOVE_ITEM OGSIMSESSION_TEST_SOURCES "${CMAKE_CURRENT_SOURCE_DIR}/tests/HeadlessSessionIntegrationTest.cpp")
    add_executable(og_sim_session_tests ${OGSIMSESSION_TEST_SOURCES} "${OGSIMSESSION_TESTS_DIR}/OgTagAliases.cpp")
    target_include_directories(og_sim_session_tests PRIVATE "${OGSIMSESSION_TESTS_DIR}")
    target_compile_definitions(og_sim_session_tests PRIVATE WITH_LOW_LEVEL_TESTS=1)
    target_link_libraries(og_sim_session_tests PRIVATE og_sim_session Catch2)
    if(TARGET og_simulation_jolt)
        target_link_libraries(og_sim_session_tests PRIVATE og_simulation_jolt)
    endif()
    if(MSVC)
        target_compile_options(og_sim_session_tests PRIVATE /bigobj /GR-)
    else()
        target_compile_options(og_sim_session_tests PRIVATE -fno-rtti)
    endif()
    if(ANDROID)
        target_link_libraries(og_sim_session_tests PRIVATE log)
    endif()
    if(EXISTS "${OGNET_CATCH2_DIR}/Catch.cmake")
        list(APPEND CMAKE_MODULE_PATH "${OGNET_CATCH2_DIR}")
        include(Catch)
        catch_discover_tests(og_sim_session_tests TEST_SPEC "[@og]" TEST_PREFIX "og_sim_session: " PROPERTIES LABELS og_sim_session)
    else()
        add_test(NAME og_sim_session_tests COMMAND og_sim_session_tests "[@og]")
        set_tests_properties(og_sim_session_tests PROPERTIES LABELS og_sim_session)
    endif()

    option(OG_SIM_SESSION_LIVE_CHECKS "Build og_sim_session_headless_tests with OG_CHECK live (OG_CHECKS_LIVE), whatever NDEBUG says" ON)
    if(TARGET og_simulation_jolt)
        get_target_property(OGSIMSESSION_OGNET_ROOT og_net SOURCE_DIR)
        add_executable(og_sim_session_headless_tests "${CMAKE_CURRENT_SOURCE_DIR}/tests/HeadlessSessionIntegrationTest.cpp"
            "${OGSIMSESSION_TESTS_DIR}/OgTagAliases.cpp")
        target_include_directories(og_sim_session_headless_tests PRIVATE "${OGSIMSESSION_TESTS_DIR}" "${OGSIMSESSION_OGNET_ROOT}/tests")
        target_compile_definitions(og_sim_session_headless_tests PRIVATE WITH_LOW_LEVEL_TESTS=1)
        if(OG_SIM_SESSION_LIVE_CHECKS)
            target_compile_definitions(og_sim_session_headless_tests PRIVATE OG_CHECKS_LIVE)
        endif()
        target_link_libraries(og_sim_session_headless_tests PRIVATE og_sim_session og_simulation_jolt Catch2)
        if(MSVC)
            target_compile_options(og_sim_session_headless_tests PRIVATE /bigobj /GR-)
        else()
            target_compile_options(og_sim_session_headless_tests PRIVATE -fno-rtti)
        endif()
        if(ANDROID)
            target_link_libraries(og_sim_session_headless_tests PRIVATE log)
        endif()
        if(EXISTS "${OGNET_CATCH2_DIR}/Catch.cmake")
            catch_discover_tests(og_sim_session_headless_tests TEST_SPEC "[@og]" TEST_PREFIX "og_sim_session: "
                PROPERTIES LABELS og_sim_session)
        else()
            add_test(NAME og_sim_session_headless_tests COMMAND og_sim_session_headless_tests "[@og]")
            set_tests_properties(og_sim_session_headless_tests PROPERTIES LABELS og_sim_session)
        endif()
        if(OG_SIM_SESSION_LIVE_CHECKS)
            add_test(NAME og_sim_session_headless_thread_check_fires
                COMMAND og_sim_session_headless_tests "HeadlessSession.StartFrameFromASecondThreadTripsTheThreadCheck")
            add_test(NAME og_sim_session_headless_netio_check_fires
                COMMAND og_sim_session_headless_tests "HeadlessSession.AServiceNetOpOffOgNetsIoThreadTripsTheExactRow")
            set_tests_properties(og_sim_session_headless_thread_check_fires og_sim_session_headless_netio_check_fires PROPERTIES
                LABELS og_sim_session
                PASS_REGULAR_EXPRESSION "OG_CHECK failed: isAllowed\\(op\\) \\([^\n]*SessionThreads\\.h:[0-9]+\\)"
                FAIL_REGULAR_EXPRESSION "did NOT fire")
        endif()
    endif()
endif()
